```jsx
<DropdownMenu trigger={<Button variant="ghost" size="icon"><Icon name="ellipsis" /></Button>} items={[{ label: 'Modifier', icon: 'pencil' }, { label: 'Envoyer la fiche PDF', icon: 'file-text' }, '-', { label: 'Désactiver', icon: 'user-x', tone: 'danger' }]} />
```
